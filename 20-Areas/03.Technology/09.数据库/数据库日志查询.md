```sql
-- 紧急：先添加几个大日志组缓解问题
ALTER DATABASE ADD LOGFILE GROUP 13 '/infodba/app/oracle/oradata/TC/redo13.log' SIZE 1G;
ALTER DATABASE ADD LOGFILE GROUP 14 '/infodba/app/oracle/oradata/TC/redo14.log' SIZE 1G;
ALTER DATABASE ADD LOGFILE GROUP 15 '/infodba/app/oracle/oradata/TC/redo15.log' SIZE 1G;

-- 切换日志让新组生效
ALTER SYSTEM SWITCH LOGFILE;

-- 检查状态
SELECT group#, sequence#, bytes/1024/1024 as mb, status FROM v$log;
```
