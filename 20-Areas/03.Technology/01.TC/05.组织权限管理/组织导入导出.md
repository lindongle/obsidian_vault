导出：
l10n_import_export -mode=export -u=infodba -p=infodba -g=dba -type=Group -file=groups -local=ZH_CN -properties=name -master
l10n_import_export -mode=export -u=infodba -p=infodba -g=dba -type=Role -file=role -properties=role_name -masterter
导入：
l10n_import_export -mode=import -u=infodba -p=infodba -g=dba -file=D:\groups_en_us.xml
l10n_import_export -mode=import -u=infodba -p=pdminfodba 1234 -g=dba -file=D:\groups_zh_CN.xml
--这种可以导出本地化值
导出：
plmxml_export -u=infodba -p=infodba -g=dba -xml_file=Group -class=Group
plmxml_export -u=infodba -p=infodba -g=dba -xml_file=User -class=User
导入：
plmxml_import -u=infodba -p=infodba -g=dba -xml_file=Group.plmxml -import_mode=overwrite
