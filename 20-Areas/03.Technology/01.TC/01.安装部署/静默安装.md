rem net use R: 自动创建共享文件夹的网络磁盘映射
rem copy 复制文件
rem /wait 等待安装程序结束，再往下执行
rem /s 静默安装
rem /qn 不显示用户界面
rem /norestart 不需要重启计算机
rem /tokens 提取注册表取得的值的第几列数据

@echo off
echo \*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*
echo 创建网络磁盘映射
echo \*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*
net use R: [\\192.168.4.11\PLMInstall](file://192.168.4.11/PLMInstall) 2\>nul
rem md C:\apps
copy R:\postprocessing\silentxml\4Cilentsilent.xml C:\\
copy R:\postprocessing\silentxml\EDAsilent.xml C:\\
rem copy R:\JreInstall\jre-8u162-windows-x64.exe C:\apps
set filename=C:\4Cilentsilent.xml
set EDAfilename=C:\4Cilentsilent.xml
set installingUser value="Administrator"
set installingUser value1="%username%"
setlocal enabledelayedexpansion
for /f "delims=" %%a in (%filename%) do (
set aa=%%a
set aa=!aa:%installingUser value%=%installingUser value1%!
echo !aa!\>\>"%filename%.tmp"
)
move "%filename%.tmp" "%filename%"

for /f "delims=" %%b in (%EDAfilename%) do (
set bb=%%b
set bb=!bb:%installingUser value%=%installingUser value1%!
echo !bb!\>\>"%EDAfilename%.tmp"
)
move "%EDAfilename%.tmp" "%EDAfilename%"

echo 检查JRE是否已安装
if exist "C:\Java" (
echo C:\Java 已存在 JRE安装取消！
goto next1
)
echo \*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*
echo 正在安装JRE.....
echo \*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*
start /WAIT R:\JreInstall\jre-8u162-windows-x64.exe INSTALLDIR=C:\Java\jre1.8.0_162_64 /s
start /WAIT R:\JreInstall\jre-8u162-windows-i586.exe INSTALLDIR=C:\Java\jre1.8.0_162_32 /s

:next1
echo 检查TC是否已安装
