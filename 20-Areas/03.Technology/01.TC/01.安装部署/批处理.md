@echo off 不显示执行的命令，只显示echo内容和执行过程及结果
@echo on 显示执行的命令

常用批处理命令：
杀掉进程
taskkill /f /im Teamcenter.exe
设置磁盘映射
net use R: [\\172.1.1.12\ClientShare](file://172.1.1.12/ClientShare) 1
设置系统环境变量
setx FMS_HOME "D:\Siemens\Teamcenter11\tccs" /m

复制文件夹
xcopy D:\tcclient\SiemensClientUpdate_Test\jar D:\Siemens\Teamcenter11\portal\plugins /E /Y /I /F

创建文件夹
md C:\Temp
删除文件夹
rd /s /q "%userprofile%\FCCCache\\
删除文件
del /f /s /q "%userprofile%\Teamcenter\\.\*"

检查文件夹存在不存在
if exist "C:\Progra~1\Java\jre7" (
echo Java已安装，自动跳过此步！
goto VC++
)

安装jdk
start /WAIT %InstallDir%\jdk-7u45-windows-x64.exe /s /norestart /v /qn INSTALLDIR=C:\Progra~1\Java\jdk1.7.0_45 AUTO_UPDATE=Disable

执行exe
start /WAIT %InstallDir%\vc++\VC_redist.x64.exe /quiet

导入注册表
reg import "D:\Siemens\Teamcenter11\InstallPackage\Siemens.reg"

永久添加path，当前窗口不生效，需单独set一遍
setx "PATH" "%PATH%;D:\Siemens\TeamcenterEDA2\eda" /m
set PATH=%PATH%D:\Siemens\TeamcenterEDA2\eda
