in a 2 tier env.
set the debug env to

set TC_DEBUG=1
set TC_POM_JOURNALLING=Y
set TC_Journalling=ON
set TC_journaling=ON
set TC_Journal_Modules=ALL
<span style='color:#FA0000'>set TC_JOURNAL=FULL</span>
<span style='color:#FA0000'>set TC_JOURNAL_PERFORMANCE_ONLY=2</span>
<span style='color:#FA0000'>set TC_SQL_DEBUG=JPT</span>
<span style='color:#FA0000'>set TC_KEEP_SYSTEM_LOG=TRUE</span>
set TC_TRACEBACK=ON
set TC_SQL_DEBUG=BJPT
set TC_DATASET_DEBUG=ON
set TC_KEEP_SYSTEM_LOG=Y
set IMAN_SDL_MAGIC=1
set IMAN_SERVER_MAGIC=2
set IMAN_SQL_JOURNAL_COUNT=100
set API_JOURNAL=FULL
set TC_JOURNAL_LINE_LIMIT=0

set UGII_CHECKING_LEVEL=1
set TC_COMMUNICATION_LOGGING=DEBUG
set TC_COMMUNICATIONS_LOGGING=DEBUG

set DEBUG_SOACAD=syslog
set TC_SOACLIENT_LOGGING=DEBUG
