C:\Windows\system32\>SET TC_DATA=D:\Siemens\tcdata

C:\Windows\system32\>D:\Siemens\tcdata\tc_profilevars

C:\Windows\system32\>cd /d D:\Siemens\Teamcenter13\pool_manager

D:\Siemens\Teamcenter13\pool_manager\>java -cp Jetty\jetty-util-9.4.26.v20200117.jar org.eclipse.jetty.util.security.Password infodba
