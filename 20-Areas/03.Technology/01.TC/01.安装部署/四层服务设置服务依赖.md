sc config "wlsvc base_domain_AdminServer" depend="OracleServiceTC11"
sc config "Teamcenter Server Manager plmtest_PoolA" depend="wlsvc base_domain_AdminServer"

sc config "Teamcenter Dispatcher Scheduler V11000.2.0.61_20180727.00" depend="OracleServiceTC11"
sc config "Teamcenter Dispatcher Module V11000.2.0.61_20180727.00" depend="Teamcenter Dispatcher Scheduler V11000.2.0.61_20180727.00"
sc config "Teamcenter DispatcherClient V11000.2.0.61_20180727.00" depend="Teamcenter Dispatcher Module V11000.2.0.61_20180727.00"
