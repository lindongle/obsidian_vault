在异地卷服务器上安装FSC服务器缓存及fundation

重启从卷fsc服务前，要把主fsc的fmsmaster文件内容手动粘贴到从卷fsc配置文件中（下面实例含传输卷，即传输卷和异地卷使用的同一个目录）。
\<?xml version="1.0" encoding="UTF-8"?\>
\<!DOCTYPE fmsworld SYSTEM "fmsmasterconfig.dtd"\>
\<fmsworld\>
\<fmsenterprise id="-1860005829" volumestate="normal"\>
\<fccdefaults\>
\<property name="FCC_CacheLocation" value="\$HOME/FCCCache\|/tmp/\$USER/FCCCache" overridable="true" /\>
\<property name="FCC_HashBlockPages" value="6144" overridable="true" /\>
\<property name="FCC_LogFile" value="\$HOME/fcc.log\|/tmp/\$USER/fcc.log" overridable="true" /\>
\<property name="FCC_MaxExtentFileSizeMegabytes" value="256" overridable="true" /\>
\<property name="FCC_MaxExtentFiles" value="11" overridable="true" /\>
\<property name="FCC_MaxReadCacheSize" value="1000M" overridable="true" /\>
\<property name="FCC_MaxWriteCacheSize" value="1000M" overridable="true" /\>
\<property name="FCC_MaximumNumberOfFilePages" value="28672" overridable="true" /\>
\<property name="FCC_MaximumNumberOfSegments" value="10688" overridable="true" /\>
\</fccdefaults\>
\<fscgroup id="mygroup"\>
 \<fsc id="FSC_PLMWEB2_Administrator" address="http://PLMWEB2:4544" ismaster="false"\>
 \<volume id="0acd5ca485f89122903b" enterpriseid="-1860005829" root="D:\Siemens\offsitevolume" priority="0" /\>
 \<transientvolume id="aa5e4cea4388d1eeeeec604d44c38cd9" enterpriseid="-1860005829" root="C:\Temp\transientVolume_infodba" priority="0" /\>
 \</fsc\>
\<fsc id="FSC_teamcentertest_tctest" address="http://teamcentertest:4544" ismaster="true"\>
\<volume id="00eb5bc5cd339089f9bb" enterpriseid="-1860005829" root="D:\Siemens\volume" priority="0" /\>
\<transientvolume id="aec943cb3d1c7217301178e87d16278c" enterpriseid="-1860005829" root="C:\Temp\transientVolume_infodba" priority="0" /\>
\</fsc\>
\<clientmap subnet="127.0.0.1" mask="0.0.0.0"\>
\<assignedfsc fscid="FSC_teamcentertest_tctest" priority="0" /\>
\<assignedfsc fscid="FSC_PLMWEB2_Administrator" priority="1" /\>
\</clientmap\>
\</fscgroup\>
\</fmsenterprise\>
\</fmsworld\>
