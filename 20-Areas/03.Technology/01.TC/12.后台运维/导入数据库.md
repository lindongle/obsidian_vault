导出：
sqlplus system/infodba@tc11
create directory backupdir as 'D:/backup';---create directory backupdir as '/app/oracle_backup';
Grant read,write on directory backupdir to public;
exit
expdp infodba/infodba@tc directory=backupdir dumpfile=infodba20221214.DMP schemas=infodba
expdp code/code@plmprd directory=backupdir dumpfile=code202006222141.DMP schemas=code

删除并导入
备份文件放到D:/backup下
sqlplus system/infodba@tc11
Drop user infodba cascade;

无法删除已连接用户
select username,sid,serial# from v\$session where username='INFODBA';
alter system kill session'9,4';

grant dba to infodba;
create directory backupdir as 'D:/backup';
Grant read,write on directory backupdir to public;
Exit
impdp infodba/infodba@tcprd directory=backupdir dumpfile='INFODBA20221214.dmp' remap_schema=infodba:infodba full=y

删除用户infodba
Drop user infodba cascade;

impdp system/infodba@tc11 directory=backdir dumpfile='sftc1020180315_backup_20180316.dmp' remap_schema=infodba:infodba full=y
