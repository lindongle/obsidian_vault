```sql
--将state列按0、10、20、30平均分配，务必执行提交动作
MERGE INTO T__MAINCFG_FORMULA tar
USING (
    SELECT
        ROWID rid,
        NVL(
            CASE MOD(ROW_NUMBER() OVER(ORDER BY ROWID)-1,4)
                WHEN 0 THEN 0
                WHEN 1 THEN 10
                WHEN 2 THEN 20
                WHEN 3 THEN 30
            END,
            0
        ) new_state
    FROM T__MAINCFG_FORMULA
) src
ON (tar.ROWID = src.rid)
WHEN MATCHED THEN
UPDATE SET tar.STATE = src.new_state;
--带条件
MERGE INTO T_MAINCFG_FORMULA t
USING (
    SELECT 
        GUID,
        CASE MOD(ROW_NUMBER() OVER(ORDER BY GUID)-1,4)
            WHEN 0 THEN 0
            WHEN 1 THEN 10
            WHEN 2 THEN 20
            WHEN 3 THEN 30
        END AS NEW_STATE
    FROM T_MAINCFG_FORMULA
    WHERE MESSAGE LIKE '%存在异常%' --截图查询条件
) src
ON (t.GUID = src.GUID)
WHEN MATCHED THEN
    UPDATE SET t.STATE = src.NEW_STATE;
```
