需要开BGS服务器访问邮箱服务器及端口；
如果SAP返回中文值，可能出现乱码，可以在GS的sap connet脚本中将中文调整为英文，其他脚本文件不变；
以上截图可以看到邮件日志，BGS-\>日志文件-\>
set SapComment \$Status
set SapTransID \$TransactionId
set SapTime \[clock format \[clock seconds\] -format "%Y-%m-%d %R"\]
set SapLogID \[string range \$SapComment 29 56\]
\#set SapDate \[clock format \[clock seconds\] -format "%Y%m%d"\]
\#puts "-----------------------------\$SapComment"

if {(\$Status eq "ERROR") \|\| (\$Status eq "UNKNOWN") \|\| (\$Status eq "UPDATED_WITH_ERROR")} {

set index 0
set Messages ""
set AllMessages ""
\#获取错误信息
<span style='color:#FA0000'>while { \$index \< \$::StatusInfo(NumberOfMessages) } {</span>
<span style='color:#FA0000'>set Messages \$::StatusInfo(Message:\$index)</span>
<span style='color:#FA0000'>append AllMessages "\$Messages \n"</span>
<span style='color:#FA0000'>incr index</span>
<span style='color:#FA0000'>}</span>
\#如果不存在邮箱，发duzijun@yapp.com
if {\[info exists ::TcData(WorkflowJob:owning_user:PersonAttribute:PA9)\]} {
set T4X_WF_Job_Owner_mail \[tpco_formatHEX16 \$::TcData(WorkflowJob:owning_user:PersonAttribute:PA9)\]
::T4X::TRANSLOG::writeCustomMappingLog \$TransactionId INFO "\* WF user's e-mail..: \>\$T4X_WF_Job_Owner_mail\<"
} else {
set T4X_WF_Job_Owner_mail "duzijun@yapp.com"
}
\#set LogDir \[T4X::TRANSLOG::getTransLogFile \$SessionLogChanne \$TransactionId\]
\#puts "LogDir --\$LogDir"
\#sending e-mail ：多个e-mail地址之间用空格分隔
set curtime \[clock seconds\]
set formattime \[clock format \$curtime -format {%Y/%m/%d-%T}\]
\## puts "SapLogID --\$SapLogID"
set from "info@yapp.com"
set addr \$T4X_WF_Job_Owner_mail
set smtp "172.20.1.134"
set port "25"
set subject "The Change Object:\$SAPId transferred \$Status !"
set message "ECM transferred \$SAPId to \$::T4S::SapInfo(SapSystem) at \$SapTime --\> \$SapComment\nDetail:\n\$AllMessages log:\n <https://localhost:11320/#transactionlog>"
\#set mailbody "ECM transferred \$SAPId to \$::T4S::SapInfo(SapSystem) at \$SapTime\n\$SapComment \n log:\n <https://PLMTEST:11320/#showlog;object_tag=$SapLogID>"
set xmailer "\$formattime ::T4S::ECM::CUSTOM::MAPPING::TC_Object2SAP_ChangeMaster {TransactionId ItemType ItemRevisionType args}"
set org "T4S at \[info hostname\]"
set ttype HTML ;# TEXT, HTML
set prio "high" ;# high, low, any
puts "subject:\n\$subject"
puts "message:\n\$message"
\#set attachments "" ;#"C:/temp/otto.txt C:/temp/willi.txt"

\#T4S 21版本参数发生变化
\# addrreceiver mail address
\# fromsender mail address
\# smtp\<DNS\|IP\|HOST\> address of the SMTP server
\# portsmpt server port (not yet supported, for future usage)
\# messageemail message text
\# orgmail header organisation
\# subjectsubject of the email
\# xmailermail header xheader
\# ttypeHTML \| PLAIN
\# prioMail priority indicaor "" (default) \| high \| low
\# argslist of file attachments (please make sure that the files are copied to the BGS before the mail is send)

\#puts "\$formattime \n \$mailfrom \n \$org \n mailbody:\$mailbody \n xmailer:\$xmailer"
set rc \[::T4X::CORE::sendMail2BGS2 \$addr \$from \$smtp \$port \$message \$org \$subject \$xmailer \$ttype \$prio\]
tpwrite "--- sendMail2BGS2 returns \$rc ---"
::T4X::TRANSLOG::writeCustomMappingLog \$TransactionId INFO "\$TransactionId---\$xmailer--- sendMail2BGS2 returns \$rc ---"
