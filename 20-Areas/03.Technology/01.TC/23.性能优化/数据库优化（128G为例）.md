**0、登录Oracle**
sqlplus /nolog；
connect / as sysdba

查询数据库关键参数：
```sql
SELECT 
    name AS param,
    CASE 
        WHEN name IN ('sga_target', 'pga_aggregate_target', 'memory_target', 'memory_max_target') 
            THEN ROUND(value/1024/1024, 2) || ' MB'
        ELSE value
    END AS display_value
FROM v$parameter
WHERE name IN (
    'sga_target',
    'pga_aggregate_target',
    'memory_target',
    'memory_max_target',
    'processes',
    'sessions',
    'open_cursors',
    'parallel_max_servers'
)
ORDER BY DECODE(name, 
    'memory_target', 1,
    'memory_max_target', 2,
    'sga_target', 3,
    'pga_aggregate_target', 4,
    'processes', 5,
    'sessions', 6,
    'open_cursors', 7,
    'parallel_max_servers', 8,
    9
);
```


**1、SGA/PGA参数调整**
服务器上如果只有数据库，且一个实例。
服务器物理内存的50%-60%给到SGA和PGA，如内存128G的，SGA和PGA参数之和为64G。
SGA和PGA的分配比例为5:1-6:1，如SGA=52G，PGA=12G
```sql
--查看表空间：
show parameter sga
show parameter pga
alter system set sga_max_size=52g scope=spfile;
alter system set sga_target=52g scope=spfile;
alter system set pga_aggregate_target =12g scope=spfile;
```

查询process和psession
```sql
-- 综合查看 sessions 和 processes 的最大值、当前值及使用率
SELECT 
    'SESSIONS' AS resource_type,
    (SELECT value FROM v$parameter WHERE name = 'sessions') AS max_limit,
    (SELECT COUNT(*) FROM v$session) AS current_used,
    (SELECT COUNT(*) FROM v$session WHERE status = 'ACTIVE') AS active_count,
    ROUND((SELECT COUNT(*) FROM v$session) / 
          NULLIF(TO_NUMBER((SELECT value FROM v$parameter WHERE name = 'sessions')), 0) * 100, 2) AS usage_pct
FROM dual
UNION ALL
SELECT 
    'PROCESSES' AS resource_type,
    (SELECT value FROM v$parameter WHERE name = 'processes') AS max_limit,
    (SELECT COUNT(*) FROM v$process) AS current_used,
    (SELECT COUNT(*) FROM v$process WHERE background = '0') AS active_count,
    ROUND((SELECT COUNT(*) FROM v$process) / 
          NULLIF(TO_NUMBER((SELECT value FROM v$parameter WHERE name = 'processes')), 0) * 100, 2) AS usage_pct
FROM dual;
```

查询哪个会话占用；
``` sql
-- 按用户名统计会话数
SELECT username, program, machine, status, COUNT(*) AS session_count
FROM v$session
WHERE type = 'USER'
GROUP BY username, program, machine, status
ORDER BY session_count DESC;
```

alter database drop logfile group 1;

alter database add logfile group 1 ('/app/oracle/oradata/plmprd/redo01.log') size 2048M reuse;

alter database drop logfile group 3;
--添加新组
alter database add logfile group 3 ('/app/oracle/oradata/plmprd/redo03.log') size 2048M reuse;

--切换当前：
alter system switch logfile;
alter system checkpoint;--将组2变为非活动

--再删除组2
alter database drop logfile group 2;
alter database add logfile group 2 ('/app/oracle/oradata/plmprd/redo02.log') size 2048M reuse;

--然后反复执行2次，将其中2个为非活动，1个为当前
alter system switch logfile;
alter system checkpoint;

select group#,sequence#,bytes,members,status from v\$log;
