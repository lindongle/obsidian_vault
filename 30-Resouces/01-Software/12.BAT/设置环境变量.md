Setx FMS_HOME "D:\Siemens\Teamcenter\OTW11\tccs" ---永久设置用户级环境变量
Setx FMS_HOME "D:\Siemens\Teamcenter\OTW11\tccs" /m --永久设置系统级环境变量
setx "PATH" "%PATH%;D:\java\jre8\bin" /m --path中添加环境变量
Set set JRE64_HOME=D:\java\jre8 --临时设置环境变量，当前窗口有效
wmic ENVIRONMENT where "name='test1'" delete --删除名称为test的用户和系统环境变量
