nginx代理逻辑  
  在http块中使用upstream{ }配置负载模式以及需要代理的地址，使用server{ }配置对外发布的地址、使用的upstream以及其他逻辑，因此要想实现在同一个nginx上代理多个服务只需要配置多个upstream{ }+server{ }。  
实现步骤  
  备份原来的nginx.conf文件，添加新的upstream{ }，修改其中的upstream以及需要进行代理的地址：  

upstream tcweb_server {
ip_hash;
server172.16.3.201:7001weight=2max_fails=2fail_timeout=900s;
server172.16.3.202:7001weight=2max_fails=2fail_timeout=900s;
}
upstream awc_server {
ip_hash;
server172.16.3.201:3000weight=2max_fails=2fail_timeout=900s;
server172.16.3.202:3000weight=2max_fails=2fail_timeout=900s;
}
添加新的server{ }，修改其中的listen、server_name、proxy_pass：  

server {
  listen 7001;
  server_name 172.16.3.205;
location / {
root html;
index index.html index.htm;
 proxy_pass <http://tcweb_server>;
proxy_connect_timeout20;
proxy_read_timeout900;
proxy_send_timeout900;
proxy_set_header Host\$host;
proxy_set_header X-Real-IP\$remote_addr;
proxy_set_header X-Forwarded-For\$proxy_add_x_forwarded_for;
}
error_page 500502503504 /50x.html;
  location =/50x.html {
    root html;
}
}
server {
  listen 3000;
  server_name 172.16.3.205;
# charset koi8-r;
# access_log logs/host.access.log main;
location / {
root html;
index index.html index.htm;
 proxy_pass <http://awc_server>;
proxy_connect_timeout20;
proxy_read_timeout900;
proxy_send_timeout900;
proxy_set_header Host\$host;
proxy_set_header X-Real-IP\$remote_addr;
proxy_set_header X-Forwarded-For\$proxy_add_x_forwarded_for;
}
  error_page 500502503504 /50x.html;
  location = /50x.html {
    root html;
}
}
