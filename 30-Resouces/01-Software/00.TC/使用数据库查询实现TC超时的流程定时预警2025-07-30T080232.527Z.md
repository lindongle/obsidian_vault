1.  使用场景：客户想指定某个流程发起后，审核人长期不执行发起钉钉提醒
2.  数据库语句实现步骤
2.1.首先自定义数据库表，用于指定流程模板名字

CREATE TABLE "INFODBA"."T_WORKFLOW_REMIND"
( "WORKFLOW_NAME" VARCHAR2(526 BYTE) NOT NULL ENABLE
)

2.2.构建任务未结束的任务视图

CREATE OR REPLACE FORCE EDITIONABLE VIEW "INFODBA"."T_TASK_INFO" ("TASKUID", "OBJECT_NAME", "OBJECT_TYPE", "PARENTTASK_RESULT", "RESPONSIBLE_USERID", "RESPONSIBLE_USERNAME", "PROCESSUID", "STARTTIME", "TEMPLATEUID") AS
select task.puid as taskuid ,task_wb.pobject_name AS object_name, task_wb.pobject_type as object_type,task.PTASK_RESULT as PARENTtask_result,
pu.puser_id as responsible_userid
,pu.puser_name as responsible_username
,task.RPARENT_PROCESSU as processuid,
task.pfnd0startdate as starttime,
t.RFND0ROOTTASKTEMPLATEU as templateuid
from PEPMTASK task,pworkspaceobject task_wb,ppom_user pu,pepmtasktemplate t where
task.PTASK_RESULT='Unset' and task_wb.puid=task.PUID
and task.rresponsible_partyu = pu.puid
and task.PSTATE_VALUE = 4
and task_wb.pobject_type in ('EPMReviewTask','EPMDoTask','EPMAcknowledgeTask')
AND t.PUID = task.RTASK_TEMPLATEU;

2.3.构建所有流程视图

CREATE OR REPLACE FORCE EDITIONABLE VIEW "INFODBA"."T_WORKFLOW_JOB_INFO" ("JOB_PUID", "JOB_OBJECT_NAME", "RROOT_TASKU") AS
select epmjob.PUID as job_puid,job_wb.pobject_name AS job_object_name,epmjob.rroot_tasku from PEPMJOB epmjob,pworkspaceobject job_wb where job_wb.puid= epmjob.PUID;

2.4构建审签未结束视图

CREATE OR REPLACE FORCE EDITIONABLE VIEW "INFODBA"."T_WORKFLOW_PERFORMSIGNOFFTASK" ("TASKUID", "TASK_RESULT", "PARENT_TASKU", "RESPONSIBLE_USERID", "RESPONSIBLE_USERNAME") AS
select task.puid as taskuid ,task.PTASK_RESULT as task_result,task.Rparent_tasku as parent_tasku
,pu.puser_id as responsible_userid
,pu.puser_name as responsible_username
from PEPMTASK task,pworkspaceobject task_wb ,psignoff signoff,pimanrelation rts,ppom_user pu
where
task.PTASK_RESULT='Unset' and task_wb.puid=task.PUID
and task_wb.pobject_type in ('EPMPerformSignoffTask')
and signoff.puid = rts.rsecondary_objectu
and task.puid = rts.rprimary_objectu
andsignoff.RFND0PERFORMERU=pu.puid;

 2.5.合并以上三个视图，构建流程汇总视图

CREATE OR REPLACE FORCE EDITIONABLE VIEW "INFODBA"."T_WORKFLOW_INFO" ("JOB_PUID", "JOB_OBJECT_NAME", "TASKUID", "OBJECT_NAME", "OBJECT_TYPE", "PARENTTASK_RESULT", "STARTTIME", "RESPONSIBLE_USERID", "RESPONSIBLE_USERNAME", "USER_RESULT", "PERFMER_USERID", "PERFMER_USERNAME", "TEMPLATEUID", "RROOT_TASKU") AS
SELECT JOBINFO.JOB_PUID,JOBINFO.JOB_OBJECT_NAME,TASKINFO.TASKUID,TASKINFO.OBJECT_NAME,TASKINFO.OBJECT_TYPE,TASKINFO.PARENTTASK_RESULT,TASKINFO.starttime + INTERVAL '8' HOUR AS STARTTIME,TASKINFO.RESPONSIBLE_USERID,TASKINFO.RESPONSIBLE_USERNAME,
PERFERMINFO.TASK_RESULT AS USER_RESULT,PERFERMINFO.RESPONSIBLE_USERID AS PERFMER_USERID,PERFERMINFO.RESPONSIBLE_USERNAME AS PERFMER_USERNAME
,TASKINFO.templateuid,JOBINFO.rroot_tasku
FROM T_WORKFLOW_JOB_INFO JOBINFO
JOIN T_TASK_INFO TASKINFO ON TASKINFO.PROCESSUID = JOBINFO.JOB_PUID
left JOIN T_WORKFLOW_PERFORMSIGNOFFTASK PERFERMINFO ON PERFERMINFO.parent_tasku = TASKINFO.TASKUID;

2.6.根据自定义的表，构建指定的流程模板视图

CREATE OR REPLACE FORCE EDITIONABLE VIEW "INFODBA"."T_WORKFLOW_TEMPALTE" ("PUID", "PTEMPLATE_NAME", "VLA_787_2", "PICON_KEY", "VLA_787_5", "VLA_787_6", "PLOCATION", "PFND0ORIGIN_UID", "VLA_787_15", "VLA_787_17", "VLA_787_18", "PFND0DECISIONSETLOV", "PFND0TASKPATHNAME", "PFND0INSTRUCTIONS", "PSTAGE", "PTEMPLATE_CLASSIFICATION", "RPARENT_TASK_TEMPLATEU", "RPARENT_TASK_TEMPLATEC", "ROBJECT_TYPEU", "ROBJECT_TYPEC", "PWAIT_FOR_ALL_REVIEWERS", "RFND0FILTERCONDITIONU", "RFND0FILTERCONDITIONC", "PSIGNOFF_QUORUM", "RFND0ROOTTASKTEMPLATEU", "RFND0ROOTTASKTEMPLATEC", "PFND0ASYNCPRIORITY", "PFND0STOREPARTICIPANTSONJOB", "PFND0EXECUTEASYNC", "PFND0REQUIRECOMPLETECONFIRM", "PSHOW_IN_PROCESS_STAGE", "PFND0INTERACTIVETASK", "VLA_787_28") AS
select t."PUID",t."PTEMPLATE_NAME",t."VLA_787_2",t."PICON_KEY",t."VLA_787_5",t."VLA_787_6",t."PLOCATION",t."PFND0ORIGIN_UID",t."VLA_787_15",t."VLA_787_17",t."VLA_787_18",t."PFND0DECISIONSETLOV",t."PFND0TASKPATHNAME",t."PFND0INSTRUCTIONS",t."PSTAGE",t."PTEMPLATE_CLASSIFICATION",t."RPARENT_TASK_TEMPLATEU",t."RPARENT_TASK_TEMPLATEC",t."ROBJECT_TYPEU",t."ROBJECT_TYPEC",t."PWAIT_FOR_ALL_REVIEWERS",t."RFND0FILTERCONDITIONU",t."RFND0FILTERCONDITIONC",t."PSIGNOFF_QUORUM",t."RFND0ROOTTASKTEMPLATEU",t."RFND0ROOTTASKTEMPLATEC",t."PFND0ASYNCPRIORITY",t."PFND0STOREPARTICIPANTSONJOB",t."PFND0EXECUTEASYNC",t."PFND0REQUIRECOMPLETECONFIRM",t."PSHOW_IN_PROCESS_STAGE",t."PFND0INTERACTIVETASK",t."VLA_787_28" from t_workflow_remind remind,pepmtasktemplate t where remind.WORKFLOW_NAME = t.PTEMPLATE_NAME and pstage = 2;

2.6.汇总成指定流程任务长期未审核视图
（时间计算自己定义，注意系统时间8小时之差）

CREATE OR REPLACE FORCE EDITIONABLE VIEW "INFODBA"."T_WORKFLOW_TIMEOUT_GENERAL" ("JOB_PUID", "JOB_OBJECT_NAME", "TASKUID", "OBJECT_NAME", "OBJECT_TYPE", "PARENTTASK_RESULT", "STARTTIME", "RESPONSIBLE_USERID", "RESPONSIBLE_USERNAME", "USER_RESULT", "PERFMER_USERID", "PERFMER_USERNAME", "TEMPLATEUID") AS
SELECT workflow."JOB_PUID",workflow."JOB_OBJECT_NAME",workflow."TASKUID",workflow."OBJECT_NAME",workflow."OBJECT_TYPE",workflow."PARENTTASK_RESULT",workflow."STARTTIME",workflow."RESPONSIBLE_USERID",workflow."RESPONSIBLE_USERNAME",workflow."USER_RESULT",workflow."PERFMER_USERID",workflow."PERFMER_USERNAME",workflow."TEMPLATEUID"
FROM T_WORKFLOW_INFO workflow,T_WORKFLOW_TEMPALTE w_template
WHERE TRUNC(workflow.starttime) = TRUNC(SYSDATE) - 0 and w_template.puid = workflow.TEMPLATEUID;
