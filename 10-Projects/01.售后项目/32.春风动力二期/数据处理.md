--配置打点括号丢失问题处理
SELECT t.*
FROM T__BOMSENDLOG t
WHERE t.TRANS_RESULTRELATION = 'S'
AND t.BATCH_NO = (
    SELECT MAX(BATCH_NO)
    FROM T__BOMSENDLOG sub
    WHERE sub.ITEM_ID = t.ITEM_ID
      AND sub.PLANT = t.PLANT
      AND sub.TRANS_RESULTRELATION = 'S'
)
AND EXISTS (
    SELECT 1
    FROM JSON_TABLE(
        t.TRANS_DATARELATION,
        '$.rows[*]' COLUMNS (
            CLAUSE VARCHAR2(1000) PATH '$.CLAUSE'
        )
    ) jt
    WHERE jt.CLAUSE IS NOT NULL
      AND INSTR(UPPER(jt.CLAUSE),'AND') > 0
      AND INSTR(UPPER(jt.CLAUSE),'OR') > 0
      AND INSTR(jt.CLAUSE,'(') = 0
)
ORDER BY t.ITEM_ID, t.PLANT;

--更新BOM类型，针对工程BOM类型，修改BOM类型，M+四位数字结尾或最后_后为4位且第4位为M。
--备份数据
CREATE TABLE PY6_TY_ITEMSREVISION_20260907 AS SELECT * FROM PY6_TY_ITEMSREVISION;
--更新MBOM
UPDATE py6_ty_itemsrevision trev
SET py6_BOMType = 'EBOM'
WHERE trev.py6_BOMType IS NULL
  AND EXISTS (
    SELECT 1
    FROM pitemrevision rev
    WHERE rev.puid = trev.puid
      AND EXISTS (
        SELECT 1
        FROM pY6_TY_MainCfgRevision crev
        WHERE crev.puid = rev.puid
      )
      AND EXISTS (
        SELECT 1
        FROM pitem itm
        WHERE itm.puid = rev.ritems_tagu
          AND NOT (
            REGEXP_LIKE(itm.pitem_id, 'M[0-9]{4}$')
            OR REGEXP_LIKE(itm.pitem_id, '_[^_]{3}M$')
          )
      )
  );
