```sql
--配置打点括号丢失问题处理
SELECT t.*
FROM T__BOMSENDLOG t
WHERE t.TRANS_RESULTRELATION = 'S'
AND t.BATCH_NO = (
    SELECT MAX(BATCH_NO)
    FROM T__BOMSENDLOG sub
    WHERE sub.ITEM_ID = t.ITEM_ID
      AND sub.PLANT = t.PLANT
      AND sub.TRANS_RESULTRELATION = 'S'
)
AND EXISTS (
    SELECT 1
    FROM JSON_TABLE(
        t.TRANS_DATARELATION,
        '$.rows[*]' COLUMNS (
            CLAUSE VARCHAR2(1000) PATH '$.CLAUSE'
        )
    ) jt
    WHERE jt.CLAUSE IS NOT NULL
      AND INSTR(UPPER(jt.CLAUSE),'AND') > 0
      AND INSTR(UPPER(jt.CLAUSE),'OR') > 0
      AND INSTR(jt.CLAUSE,'(') = 0
)
ORDER BY t.ITEM_ID, t.PLANT;
```
